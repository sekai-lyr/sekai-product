
    // Sekai: resolve collisions in the final, scaled word boxes before drawing.
    const boxes = placements.map(p => ({
        left: p.x - p.width * (p.scale - 1) / 2,
        right: p.x + p.width * (p.scale + 1) / 2,
        top: p.y - fontPx * p.scale * 1.1,
        bottom: p.y + fontPx * p.scale * .3,
    }));
    const collides = boxes.some((a, i) => boxes.slice(i + 1).some(b =>
        a.left < b.right + 10 && a.right + 10 > b.left && a.top < b.bottom + 10 && a.bottom + 10 > b.top));
    if (collides) {
        const ordered = [...placements].sort((a, b) => a.wordIndex - b.wordIndex || a.fragmentStartInWord - b.fragmentStartInWord);
        const rows: Array<{ items: typeof placements; width: number; height: number }> = [];
        for (const p of ordered) {
            const w = p.width * p.scale + 22;
            let row = rows[rows.length - 1];
            if (!row || row.width + w > maxWidth) { row = { items: [], width: 0, height: 0 }; rows.push(row); }
            row.items.push(p); row.width += w; row.height = Math.max(row.height, fontPx * p.scale * 1.65 + 18);
        }
        const total = rows.reduce((sum, row) => sum + row.height, 0);
        let y = -total / 2;
        for (const row of rows) {
            let x = -row.width / 2;
            for (const p of row.items) {
                p.x = x + p.width * (p.scale - 1) / 2 + 11;
                p.y = y + row.height * .72; p.rotate = 0; p.passedRotate = 0;
                p.passedDriftX = 0; p.passedDriftY = 0;
                x += p.width * p.scale + 22;
            }
            y += row.height;
        }
    }
