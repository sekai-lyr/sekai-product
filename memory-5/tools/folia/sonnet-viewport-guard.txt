
            // Sekai adaptation: fit actual glyph bounds after camera/transition transforms.
            // Decorations may leave the frame; meaningful foreground words must not.
            const foreground = scene.shots[visibleShotIndex]?.segments
                .filter(segment => segment.role !== 'decoration')
                .flatMap(segment => segment.glyphs.map(glyph => glyph.display)) ?? [];
            const measureForeground = () => {
                let left = Infinity, top = Infinity, right = -Infinity, bottom = -Infinity;
                for (const glyph of foreground) {
                    const bounds = glyph.getBounds();
                    if (!Number.isFinite(bounds.x + bounds.y + bounds.width + bounds.height)) continue;
                    left = Math.min(left, bounds.x); top = Math.min(top, bounds.y);
                    right = Math.max(right, bounds.x + bounds.width); bottom = Math.max(bottom, bounds.y + bounds.height);
                }
                return { left, top, right, bottom, width: right - left, height: bottom - top };
            };
            let bounds = measureForeground();
            if (bounds.width > 0 && bounds.height > 0) {
                const mobileViewport = window.matchMedia('(max-width:900px), (max-height:600px) and (max-width:1100px)').matches && !!document.documentElement.style.getPropertyValue('--performance-height');
                const paddingX = Math.max(mobileViewport ? 12 : 24, width * .045);
                const safeTop = mobileViewport ? Math.max(12, height * .06) : Math.min(height * .22, width < 600 ? 160 : 110);
                const safeBottom = height - (mobileViewport ? Math.max(12, height * .06) : Math.min(height * .3, width < 600 ? 220 : 140));
                const fit = Math.min(1, (width - paddingX * 2) / bounds.width, (safeBottom - safeTop) / bounds.height);
                scene.container.scale.set(scene.container.scale.x * fit, scene.container.scale.y * fit);
                bounds = measureForeground();
                const dx = bounds.left < paddingX ? paddingX - bounds.left : bounds.right > width - paddingX ? width - paddingX - bounds.right : 0;
                const dy = bounds.top < safeTop ? safeTop - bounds.top : bounds.bottom > safeBottom ? safeBottom - bounds.bottom : 0;
                scene.container.x += dx; scene.container.y += dy;
                const fitted = measureForeground();
                this.options.host.dataset.sekaiViewport = JSON.stringify({
                    left: fitted.left, top: fitted.top, right: fitted.right, bottom: fitted.bottom,
                    safeLeft: paddingX, safeTop, safeRight: width - paddingX, safeBottom,
                });
            }
